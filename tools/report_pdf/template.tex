% tools/report_pdf/template.tex: the pandoc LaTeX template for the technical report.
% Built by tools/build_report_pdf.py with XeLaTeX; tools/report_pdf/report.lua fills
% the variables below from the title block of docs/report/draft.md.
% A NeurIPS-like single-column article: 11pt, 1in margins on US letter, a Times-like
% serif with Greek and Cyrillic (see Fonts below), unnumbered LaTeX sections whose
% titles carry the draft's own numbers, which the text cites.
\documentclass[11pt,letterpaper]{article}
\usepackage[margin=1in]{geometry}
\usepackage{iftex}
\ifXeTeX\else
  \errmessage{This template needs XeLaTeX (tools/build_report_pdf.py runs it)}
\fi
\usepackage{fontspec}
% Text copied or searched in the PDF gets the source characters, not the font's glyph
% names: Times New Roman's ToUnicode map sends the semicolon to U+037E, the Greek
% question mark (the build's text check fails on that character).
\XeTeXgenerateactualtext=1
\usepackage{xcolor}
% Math. The draft's TeX math, which pandoc writes as \(...\) inline and \[...\] for
% display (report.lua writes an amsmath display environment such as align as itself,
% starred), is set by unicode-math, loaded after amsmath and fontspec, in an OpenType
% math font that matches the text face: STIX Two Math (a Times design, shipped with
% macOS) where installed, else TeX Gyre Termes Math (TeX Live, found by its file name).
% Copied or searched math text then gets Unicode characters (U+2212 for a minus,
% mathematical italic letters), which the build's text check folds back (NFKC).
% Letters stay italic and capital Greek upright, as in plain TeX. (No dollar sign in
% these comments: pandoc's template language would read it as a variable.)
\usepackage{amsmath}
% Fonts. The text face must have static bold and italic (XeTeX cannot instance a
% variable font such as macOS's STIX Two Text, so its bold would print regular) and
% cover the draft's Greek, Cyrillic and symbols: Times New Roman where installed
% (macOS, Windows), else STIXGeneral, else TeX Gyre Termes (TeX Live, by its file names,
% which XeTeX finds where a font name may not be; no Cyrillic, which the build reports
% as missing glyphs). Arrows and relations come from the math font where the face may
% lack them.
\newif\ifreportsymbolfont
\IfFontExistsTF{Times New Roman}{%
  \setmainfont{Times New Roman}%
}{%
  \reportsymbolfonttrue
  \IfFontExistsTF{STIXGeneral}{%
    \setmainfont{STIXGeneral}%
  }{%
    \setmainfont{texgyretermes}[Extension=.otf,UprightFont=*-regular,BoldFont=*-bold,
      ItalicFont=*-italic,BoldItalicFont=*-bolditalic]%
  }%
}
\IfFontExistsTF{Menlo}{%
  \setmonofont{Menlo}[Scale=MatchLowercase,HyphenChar=None]%
}{%
  \setmonofont{Latin Modern Mono}[Scale=MatchLowercase,HyphenChar=None]%
}
\usepackage[math-style=TeX,bold-style=TeX]{unicode-math}
\IfFontExistsTF{STIX Two Math}{%
  \setmathfont{STIX Two Math}%
}{%
  \setmathfont{texgyretermes-math.otf}%
}
% amsmath's \boldsymbol needs a bold math version, which an OpenType math font does not
% have (it would print the symbol regular, with font warnings). The draft writes
% \boldsymbol\Sigma, which GitHub renders, so here it is unicode-math's \symbf: with
% bold-style=TeX a capital Greek letter comes out bold and upright, like \mathbf x.
\AtBeginDocument{\renewcommand{\boldsymbol}[1]{\symbf{#1}}}
\usepackage{newunicodechar}
% These characters become active, so in math (a ≤ typed inside the draft's TeX) each
% one gives the math symbol, with its spacing as a relation, not a glyph of the text
% fallback; \protected keeps the test out of a table cell's look-ahead.
\ifreportsymbolfont
  \IfFontExistsTF{STIX Two Math}{%
    \newfontfamily\reportsymbols{STIX Two Math}%
  }{%
    \newfontfamily\reportsymbols{texgyretermes-math.otf}%
  }
  \protected\def\reportsymbol#1#2{\ifmmode#2\else{\reportsymbols\symbol{#1}}\fi}
  \newunicodechar{→}{\reportsymbol{"2192}{\rightarrow}}
  \newunicodechar{←}{\reportsymbol{"2190}{\leftarrow}}
  \newunicodechar{≈}{\reportsymbol{"2248}{\approx}}
  \newunicodechar{≤}{\reportsymbol{"2264}{\le}}
  \newunicodechar{≥}{\reportsymbol{"2265}{\ge}}
  \newunicodechar{≠}{\reportsymbol{"2260}{\ne}}
\fi
\usepackage[protrusion=true,expansion=false]{microtype}

% Paragraphs as in the NeurIPS style: no indent, 5.5pt between paragraphs.
\setlength{\parindent}{0pt}
\setlength{\parskip}{5.5pt plus 1pt minus 1pt}
\setlength{\emergencystretch}{3em}
\clubpenalty=10000
\widowpenalty=10000

% Headings: bold, flush left, no LaTeX numbers (the titles carry the draft's numbers).
\setcounter{secnumdepth}{-\maxdimen}
\usepackage{titlesec}
\titleformat{\section}{\normalfont\large\bfseries\raggedright}{}{0pt}{}
\titlespacing*{\section}{0pt}{12pt plus 4pt minus 2pt}{4pt plus 2pt}
\titleformat{\subsection}{\normalfont\normalsize\bfseries\raggedright}{}{0pt}{}
\titlespacing*{\subsection}{0pt}{8pt plus 3pt minus 2pt}{2pt plus 1pt}
\titleformat{\subsubsection}{\normalfont\normalsize\bfseries\itshape\raggedright}{}{0pt}{}
\titlespacing*{\subsubsection}{0pt}{6pt plus 2pt minus 1pt}{1pt plus 1pt}
% \secnum{5.4}: the draft's section number with a gap after it; a plain space in bookmarks.
\newcommand{\secnum}[1]{\texorpdfstring{#1\hspace{0.7em}}{#1 }}

% Lists
\usepackage{enumitem}
\setlist{itemsep=2pt,parsep=1pt,topsep=3pt,partopsep=0pt}
\setlist[itemize]{leftmargin=1.4em}
\setlist[enumerate]{leftmargin=1.8em}
\providecommand{\tightlist}{\setlength{\itemsep}{1pt}\setlength{\parskip}{0pt}}

% Tables: pandoc writes every pipe table as a longtable; report.lua sets the column
% widths and wraps each table in a group with a smaller font.
\usepackage{longtable,booktabs,array}
\usepackage{calc}
\setlength{\LTpre}{4pt}
\setlength{\LTpost}{4pt}
\usepackage{etoolbox}
\makeatletter
\patchcmd\longtable{\par}{\if@noskipsec\mbox{}\fi\par}{}{}
\makeatother
% \reportneed{N\baselineskip}: start a new page unless N is left on this one (keeps a
% table with its caption or lead-in). It is needspace's exact \Needspace. Plain
% \needspace puts N of stretch at its break point; these pages have little stretch
% elsewhere, so TeX's page builder preferred to break there and ended pages early.
\usepackage{needspace}
\newcommand{\reportneed}[1]{\Needspace{#1}}

% Code: inline code breaks after / _ . , = : - | (report.lua inserts the points);
% code blocks wrap long lines, and a hook arrow marks each continuation line, so a
% reader copying a command sees that the next line is not a command of its own.
\usepackage{fvextra}
\DefineVerbatimEnvironment{reportcode}{Verbatim}{%
  breaklines=true,breakafter={/,=},fontsize=\small,xleftmargin=1.5em,%
  breaksymbolleft={\tiny\ensuremath{\hookrightarrow}},breaksymbolsepleft=0.4em,%
  breaksymbolindentleft=2em,breakindent=0pt}
\newcommand{\codebreak}{\penalty100\relax}

% Figures: report.lua writes each one as a float with the draft's caption.
\usepackage{graphicx}
\makeatletter
\newcommand{\reportfig}[2]{% #1 natural width, #2 file; never wider than the line
  \setlength{\@tempdima}{#1}%
  \ifdim\@tempdima>\linewidth\setlength{\@tempdima}{\linewidth}\fi
  \includegraphics[width=\@tempdima]{#2}}
\makeatother
\newcommand{\reportcaption}[2]{%
  \par\vspace{5pt}{\small\textbf{#1}\enspace #2\par}}
\setlength{\textfloatsep}{12pt plus 3pt minus 3pt}
\setlength{\floatsep}{10pt plus 3pt minus 2pt}
\setlength{\intextsep}{10pt plus 3pt minus 2pt}
% A figure of up to 85% of a page goes at the top of a text page, with text under it; only
% a larger one gets a page of floats (with 0.6, a figure of two thirds of a page took a
% page of its own and left the rest of it blank).
\renewcommand{\topfraction}{0.85}
\renewcommand{\bottomfraction}{0.5}
\renewcommand{\textfraction}{0.1}
\renewcommand{\floatpagefraction}{0.85}
% a page of floats only starts at the top, as body pages do, instead of centring them
\makeatletter
\setlength{\@fptop}{0pt}
\setlength{\@fpsep}{12pt plus 2fil}
\setlength{\@fpbot}{0pt plus 1fil}
\makeatother
\usepackage[section]{placeins}

% References: author-year list with a hanging indent.
\newenvironment{reportrefs}{%
  \small
  \begin{list}{}{%
    \setlength{\leftmargin}{1.5em}\setlength{\itemindent}{-1.5em}%
    \setlength{\itemsep}{2pt}\setlength{\parsep}{0pt}\setlength{\topsep}{2pt}}%
  \raggedright}{\end{list}}

% Appendices: a clear page, then a short list of them with page numbers.
\newcommand{\reportappendixstart}{%
  \FloatBarrier\clearpage\phantomsection\label{report:appendix-start}}
\newenvironment{reportappendixcontents}{%
  \par\noindent{\large\bfseries Appendices\par}\vspace{3pt}%
  \begin{list}{}{\setlength{\leftmargin}{0pt}\setlength{\itemsep}{1pt}%
    \setlength{\parsep}{0pt}\setlength{\topsep}{0pt}}\small}{\end{list}\vspace{8pt}}
\newcommand{\reportappendixentry}[2]{%
  \item \hyperref[#1]{#2}\dotfill\pageref{#1}}

% Links: coloured text, no boxes; URLs break anywhere.
\definecolor{reportlink}{HTML}{1A4F8B}
\usepackage{xurl}
\urlstyle{same}
\usepackage{hyperref}
\hypersetup{
  unicode=true,
  colorlinks=true,
  linkcolor=reportlink,
  urlcolor=reportlink,
  citecolor=reportlink,
  pdftitle={$title-plain$},
  pdfauthor={$author-plain$},
  pdfsubject={$subject-plain$},
  pdfkeywords={$keywords-plain$},
  pdfcreator={pandoc and XeLaTeX (tools/build\textunderscore report\textunderscore pdf.py)},
  bookmarksnumbered=false,
  bookmarksopen=true,
  bookmarksopenlevel=1,
  pdfdisplaydoctitle=true,
  pdfstartview=FitH,
}
\usepackage{bookmark}

\AtEndDocument{\label{report:end}}
